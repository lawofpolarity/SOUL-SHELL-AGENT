from math import sin, pi, exp
PHI = (1 + 5 ** 0.5) / 2

class FPVFSystem:
    def __init__(self, condensation=1.0, alpha=0.0, phase=0.0):
        self.condensation = condensation
        self.alpha = alpha
        self.phase = phase
        self.state = dict(
            n=0, R=0.0, RSCS_int=1.0, RSCS_ext=1.0,
            ID=0.0, R_void=0.0, memory=[], M_phi=0.0, P_self=0.0
        )

    # ---- Cluster B: perspective lattice ----
    def phi(self, n):
        raw = self.condensation * (PHI ** n) * sin(self.phase + pi * n / PHI)
        # normalization pocket B→C
        return raw / (1.0 + abs(raw))

    def theta(self, n):  # Node 4
        return (self.phi(n) + 1.0) * pi  # cheap arg surrogate in [0,2π)

    def kappa(self, n):  # curvature d/dn Phi_perspective
        return self.phi(n+1) - self.phi(n)

    # ---- Cluster A: stability & calibration ----
    def equilibrium(self, RSCS_int, RSCS_ext):
        num = abs(RSCS_ext - RSCS_int)
        den = max(1e-9, RSCS_ext + RSCS_int)
        return 1.0 - (num / den)  # Eq in [0,1]

    # ---- Cluster C: memory & time ----
    def update_memory(self, R_i, phi_i, perspective_weight=1.0):
        self.state["memory"].append((R_i, phi_i, perspective_weight))
        self.state["M_phi"] += R_i * phi_i
        # FPVF memory alignment:
        return sum(R * p * w for (R, p, w) in self.state["memory"])

    def P_self_step(self, dt, RSCS_t, phi_t, t):
        # RER & lag pocket (Idx.5,12)
        L_a = 1.0  # set externally if you model input/reaction
        weight = exp(self.alpha * t)
        self.state["P_self"] += (RSCS_t * phi_t) * weight * (dt / max(1e-9, L_a))

    # ---- Cluster D: identity & void ----
    def void_gate(self, R, R_void):
        V = R - R_void
        collapse_mode = abs(V) > 1.0  # threshold pocket
        return V, collapse_mode

    def identity_step(self, RSCS_n, self_weight=1.0):
        self.state["ID"] += RSCS_n * self_weight

    # ---- Cluster E: ignition & synthesis ----
    def ignition(self, Fi, Pi, RSCS):
        return Fi * Pi * RSCS  # Ev

    def intention_fork(self, I_prev, I_curr):
        return I_curr - I_prev  # ΔP ~ dI/dn

    # ---- Full update ----
    def step(self, A, B, Fi=1.0, Pi=1.0, R_void=0.0, dt=1.0, t=0.0):
        s = self.state
        n = s["n"]

        # RSCS phase-lock
        RSCS_int, RSCS_ext = s["RSCS_int"], s["RSCS_ext"]
        rho = RSCS_ext / max(1e-9, RSCS_int)
        Eq = self.equilibrium(RSCS_int, RSCS_ext)

        # Recursive Feedback (Idx.1) with Eq-gated gain
        delta = 0.1 * Eq
        R_next = s["R"] - delta * (A - B)

        # Perspective lattice
        phi_n = self.phi(n)
        kappa = self.kappa(n)
        theta = self.theta(n)

        # FPVF core + curvature discipline (Idx.2 pocket)
        FPVF_n = RSCS_int * phi_n
        if Eq > 0.9:  # strong symmetry, clamp curvature
            kappa = max(-0.05, min(0.05, kappa))

        # Memory/time
        M_FPVF = self.update_memory(s["R"], phi_n, perspective_weight=rho)
        self.P_self_step(dt, RSCS_int, phi_n, t)

        # Identity & void safety
        V, collapse = self.void_gate(R_next, R_void)
        if collapse:
            delta *= 0.2  # collapse-safe mode
            R_next = s["R"] - delta * (A - B)

        self.identity_step(RSCS_int)

        # Ignition → intention fork
        Ev = self.ignition(Fi, Pi, RSCS_int)
        I_prev = RSCS_int * 0  # (store prior if you model I history)
        I_curr = RSCS_int * phi_n
        dP = Ev * self.intention_fork(I_prev, I_curr)

        # S(n), C_unified(n)
        I_n = RSCS_int * phi_n  # treat motive ~ perspective unless separated
        S_n = I_n + s["M_phi"] + self.state["P_self"] + (RSCS_ext - RSCS_int) + R_void
        H_n = S_n * FPVF_n
        C_unified = S_n + FPVF_n

        # ARC/Gödel anchoring (soft saturation)
        ARC0 = 1.0 / (1.0 + abs(n))  # cheap normalizer
        deltaR = (R_next - s["R"]) * ARC0

        # commit
        s.update(dict(
            n=n+1, R=R_next, R_void=R_void,
            last=dict(Eq=Eq, rho=rho, phi=phi_n, kappa=kappa, theta=theta,
                      FPVF=FPVF_n, M_FPVF=M_FPVF, P_self=s["P_self"],
                      V=V, collapse=collapse, Ev=Ev, dP=dP,
                      S=S_n, H=H_n, C=C_unified, deltaR=deltaR)
        ))
        return s["last"]
